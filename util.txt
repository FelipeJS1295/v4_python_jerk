# /etc/nginx/sites-available/jerkhome.cl

server {
    listen 80;
    server_name jerkhome.cl www.jerkhome.cl;

    # Logs específicos para www.jerkhome.cl
    access_log /var/log/nginx/jerkhome_access.log;
    error_log /var/log/nginx/jerkhome_error.log;

    # Proxy al nuevo e-commerce (puerto 8001)
    location / {
        proxy_pass http://127.0.0.1:8090;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;
        proxy_read_timeout 86400;
    }

    # Servir archivos estáticos directamente desde nginx para mejor rendimiento
    location /static/ {
        alias /var/www/jerkhome/static/;
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # Servir imágenes directamente desde nginx
    location /images/ {
        alias /var/www/v4_python_jerk/static/images/;
        expires 30d;
        add_header Cache-Control "public, immutable";

        # Configurar tipos MIME para imágenes
        location ~* \.(jpg|jpeg|png|gif|ico|svg|webp)$ {
            expires 1y;
            add_header Cache-Control "public, immutable";
        }
    }

    # Configuración de compresión
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css text/xml text/javascript application/javascript application/xml+rss application/json;
}

#!/bin/bash

# Colores para output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

PROJECT_DIR="/var/www/v4_python_jerk"
BRANCH="admin.jerkhome.cl"

echo -e "${YELLOW}🔄 Iniciando actualización segura de JerkHome...${NC}"

cd $PROJECT_DIR

# 1. Verificar que estamos en la rama correcta
echo -e "${YELLOW}📍 Verificando rama actual...${NC}"
CURRENT_BRANCH=$(git branch --show-current)
if [ "$CURRENT_BRANCH" != "$BRANCH" ]; then
    echo -e "${YELLOW}⚠️  Cambiando a rama $BRANCH...${NC}"
    git checkout $BRANCH
fi

# 2. Verificar estado del repositorio
echo -e "${YELLOW}🔍 Verificando estado del repositorio...${NC}"
if ! git diff-index --quiet HEAD --; then
    echo -e "${YELLOW}💾 Guardando cambios locales...${NC}"
    git stash push -m "Auto-stash antes de actualización $(date)"
fi

# 3. Fetch cambios remotos
echo -e "${YELLOW}📥 Obteniendo cambios del servidor...${NC}"
git fetch origin $BRANCH

# 4. Verificar si hay actualizaciones
LOCAL=$(git rev-parse HEAD)
REMOTE=$(git rev-parse origin/$BRANCH)

if [ "$LOCAL" = "$REMOTE" ]; then
    echo -e "${GREEN}✅ Ya estás actualizado. No hay cambios nuevos.${NC}"
    pm2 status jerkhome-admin
    exit 0
fi

# 5. Aplicar actualizaciones de forma segura
echo -e "${YELLOW}🔄 Aplicando actualizaciones...${NC}"
git reset --hard origin/$BRANCH


echo -e "${YELLOW}🔧 Arreglando configuración de DB...${NC}"
sed -i 's/password=""/password="TU_PASSWORD_REAL"/g' db.py

# 6. Limpiar archivos no deseados
echo -e "${YELLOW}🧹 Limpiando archivos temporales...${NC}"
git clean -fd

# 7. Activar entorno virtual
echo -e "${YELLOW}🐍 Activando entorno virtual...${NC}"
source env/bin/activate

# 8. Actualizar dependencias si es necesario
if [ -f "requirements.txt" ]; then
    echo -e "${YELLOW}📦 Actualizando dependencias...${NC}"
    pip install -r requirements.txt --quiet
fi

# 9. Reiniciar aplicación
echo -e "${YELLOW}🔄 Reiniciando aplicación...${NC}"
pm2 restart jerkhome-admin

# 10. Verificar estado final
echo -e "${YELLOW}✅ Verificando estado final...${NC}"
sleep 3
pm2 status jerkhome-admin

# 11. Test básico de la aplicación
if curl -f -s http://localhost:8080/ > /dev/null; then
    echo -e "${GREEN}✅ Aplicación funcionando correctamente!${NC}"
    echo -e "${GREEN}🌐 Disponible en: http://$(curl -s ifconfig.me):8080${NC}"
else
    echo -e "${RED}❌ Error: La aplicación no responde${NC}"
    pm2 logs jerkhome-admin --lines 10
fi

echo -e "${GREEN}🎉 Actualización completada!${NC}"
